Solution

根号做法，目前最优解。

记位数为 xxx 的数的个数为 g(x)=9×10x−1g(x)=9\times 10^{x-1}g(x)=9×10x−1

显然要分类讨论 f(l)f(l)f(l) 和 f(r)f(r)f(r) 的关系。

  1. f(l)=f(r)f(l)=f(r)f(l)=f(r)：

    答案为 g(l)−nf(l)+1g(l)-\dfrac{n}{f(l)}+1g(l)−f(l)n​+1，还需要满足 f(l)∣nf(l)|nf(l)∣n，可以通过枚举因数做到 O(n)O(\sqrt n)O(n​)。因为因数个数不是瓶颈所以可以放心使用快速幂。

  2. f(l)+1=f(r)f(l)+1=f(r)f(l)+1=f(r)：

    考虑列方程 x×f(l)+y×(f(l)+1)=kx\times f(l)+y\times (f(l)+1)=kx×f(l)+y×(f(l)+1)=k 且 x,y>0x,y>0x,y>0。

    化简后有 ⌊kf(l)⌋<x+y≤⌊k−1f(l)⌋\lfloor\dfrac{k}{f(l)}\rfloor <x+y\le \lfloor\dfrac{k-1}{f(l)}\rfloor⌊f(l)k​⌋<x+y≤⌊f(l)k−1​⌋。对于 f(l)≥8f(l)\ge 8f(l)≥8 的情况没有 x,yx,yx,y 的限制，容易想到用整除分块去做。否则可以跟下面一种情况一起做。

  3. f(l)+1<f(r)f(l)+1<f(r)f(l)+1<f(r)：

    中间的数一定要取，根据 nnn 的范围可以知道 f(r)≤8f(r)\le 8f(r)≤8。那么可以直接枚举 f(l),f(r)f(l),f(r)f(l),f(r)，然后求解二元一次方程即可。注意两个系数存在上下界。

最后的复杂度就是 O(n+log⁡3n)O(\sqrt n+\log ^3 n)O(n​+log3n)。

const int mod=1e9+7;
ll power(ll x,int y=mod-2,int p=mod) {
	ll sum=1;x%=p;
	while (y) {
		if (y&1) sum=sum*x%p;
		x=x*x%p;y>>=1;
	}
	return sum;
}
int gcd(int x,int y) {return !y?x:gcd(y,x%y);}
void exgcd(int a,int b,int &x,int &y) {
	if (!b) return x=1,y=0,void();
	else exgcd(b,a%b,y,x),y-=a/b*x;
}
int n,ans;
void add(int &x,int y) {x=(x+y)%mod;}
int F(int x) {
	return power(10,x-1)*9%mod;
}
void calc(int x) {
	int res=F(x);
	if (x<=8&&res*x<n) ;
	else add(ans,res-n/x+1);
}
signed main(void){
	int i,x,y,j;
	read(n);
	for (i=1;i*i<=n;i++) if ((n%i)==0) {
		calc(i);
		if (i*i<n) calc(n/i);
	} 
	int res=ans;
	for (i=1;i<=8;i++) {
		int res=0;
		for (j=i+1;j<=8;j++) {
			if (res+i+j>n) break;
			int tmp1=F(i)-1,tmp2=F(j)-1;
			int a=i,b=j,c=n-res-i-j,d=gcd(a,b);
			res+=F(j)*j;
			if (c%d) continue;
			exgcd(a,b,x,y);
			a/=d,b/=d,c/=d;
			x*=c,y*=c;
			int L=max(ceil(1.0*(-x)/b),ceil(1.0*(y-tmp2)/a));
			int R=min(floor(1.0*(y)/a),floor(1.0*(tmp1-x)/b));
			if (L<=R) ans+=R-L+1;
		}
	}
	
	int l,r;
	for (l=9;l<=n;l=r+1) {
		r=min(n,n/(n/l));
		add(ans,mod-(r-l+1)*(n/l)%mod);
	}
	for (l=8;l<n;l=r+1) {
		r=min(n-1,(n-1)/((n-1)/l));
		add(ans,(r-l+1)*((n-1)/l));
	}
	printf("%lld\n",ans%mod);
	return 0;
}
