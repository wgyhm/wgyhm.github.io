Solution

水题不会做了，警戒！

题目就是想让我们把一张图分为至少 666 个部分（显然 666 个部分更优），使得每个部分的点只和自己部分以及相邻部分的点有边。

我们直接把第 111 部分设为 111 号点，第 666 部分设为 222 号点，与 111 相邻的点显然要在第 222 部分，与 222 相邻的点要在第 555 部分。

同样的，与第 222 部分相邻的点要在第 333 部分，与第 555 部分相邻的点一定要在第 444 部分。而现在的问题转化为如何分配还没有分配的点，使得边数最多。

令 did_idi​ 表示当前第 iii 部分每个部分的点的数量。考虑新加入每个部分的贡献。第 2,3,4,52,3,4,52,3,4,5 部分分别有 d2×d3,d2×d3×d4,d3×d4×d5,d4×d5d_2\times d_3,d_2\times d_3\times d_4,d_3\times d_4\times d_5,d_4\times d_5d2​×d3​,d2​×d3​×d4​,d3​×d4​×d5​,d4​×d5​，显然加入第 3,43,43,4 部分的贡献更大的那一个。而且这样贪心是对的，无论加入 3,43,43,4 都不会对之后加入的点的贡献比较有变化。

复杂度 O(n+m)O(n+m)O(n+m)。

最后统计边数 −m-m−m 即可，不需要像其他题解一样容斥很麻烦。

Tp void read(T &x){
    int f=1;x=0;char c=getchar();
    while (c<'0'||c>'9') {if (c=='-') f=-1;c=getchar();}
    while (c>='0'&&c<='9') {x=x*10+c-'0';c=getchar();}
    x*=f;
}

	#define gdb(...) _debug((char*)#__VA_ARGS__,__VA_ARGS__)
}using namespace Debug;

const int mod=1e9+7;
int power(int x,int y=mod-2) {
	int sum=1;
	while (y) {
		if (y&1) sum=sum*x%mod;
		x=x*x%mod;y>>=1;
	}
	return sum;
}
int n,m;
vector<int>to[maxn];
int ans;
int d[7],c[maxn];
signed main(void){
	int i,j,k,l,x,y;
    read(n);read(m);
    for (i=1;i<=m;i++) {
        read(x),read(y);
        to[x].push_back(y);
        to[y].push_back(x);
    }
    d[1]++;d[6]++;
    for (auto y:to[1]) d[2]++,c[y]=2;
    for (auto y:to[2]) d[5]++,c[y]=5;
    for (i=3;i<=n;i++) if (!c[i]) {
        int fl=0;
        for (auto y:to[i]) {
            if (c[y]==2) fl=1;
            else if (c[y]==5) fl=2; 
        }
        if (fl==1) d[3]++,c[i]=3;
        else if (fl==2) d[4]++,c[i]=4;
        else {
            if (d[2]>d[5]) d[3]++,c[i]=3;
            else d[4]++,c[i]=4;
        }
    }
    for (i=1;i<=6;i++) ans+=(d[i]*(d[i]-1)/2);
    for (i=1;i<=5;i++) ans+=d[i]*d[i+1];
    printf("%lld\n",ans-m);
	return 0;
}
