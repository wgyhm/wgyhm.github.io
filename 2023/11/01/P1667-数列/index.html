自己推了半天感觉什么都没搞出来。看了题解以后发现是个傻逼题。

**这种题目一般是让我们把操作等价为一个更简单的操作。**比如 NOIP2021 差分转化到差分数组上就是交换相邻两个数的值。

在这道题目上面就是转化为交换 sl−1,srs_{l-1},s_rsl−1​,sr​ 的值。

后面判无解什么的根据题目来就好了。

    int f=1;x=0;char c=getchar();
    while (c<'0'||c>'9') {if (c=='-') f=-1;c=getchar();}
    while (c>='0'&&c<='9') {x=x*10+c-'0';c=getchar();}
    x*=f;
}

const int mod=1e9+7;
int power(int x,int y=mod-2) {
	int sum=1;
	while (y) {
		if (y&1) sum=sum*x%mod;
		x=x*x%mod;y>>=1;
	}
	return sum;
}
int n,a[maxn],s[maxn],ans,g[maxn];
signed main(void){

	for (i=1;i<=n;i++) read(a[i]),s[i]=s[i-1]+a[i];
	for (i=1;i<=n;i++) g[i]=s[i];
	sort(g+1,g+n+1);
	int tot=unique(g+1,g+n+1)-g-1;
	if (g[1]<=0) return puts("-1"),0;
	if (tot!=n) return puts("-1"),0;

	if (s[n]!=n) return puts("-1"),0;
