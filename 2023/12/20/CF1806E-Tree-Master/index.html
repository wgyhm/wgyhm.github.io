Solution

水群的时候发现的很有意思的题，一眼只能想到树分块和莫队。

首先考虑根号分治，对深度是没有前途的，转而对每层节点数分治：

  • 当前层节点数 x<nx < \sqrt nx<n​，总共有 (x2)=O(x2)\dbinom{x}{2}=O(x^2)(2x​)=O(x2) 种本质不同的询问，所以 <n<\sqrt n <n​ 的总的本质不同的询问有 ∑xi2≤(∑xi)×(max⁡xi)≤O(nn)\sum x_i^2\le (\sum x_i)\times (\max x_i )\le O(n\sqrt n)∑xi2​≤(∑xi​)×(maxxi​)≤O(nn​)。
  • 对于 x≥nx\ge \sqrt nx≥n​，这样的层只有 O(n)O(\sqrt n )O(n​) 个，每次询问查询一次就是 O(qn)O(q\sqrt n)O(qn​)。

为了避免本质相同的询问带来的影响，考虑记忆化搜索。只需要把节点数 <n<\sqrt n<n​ 的记下来即可，可以避免使用哈希表或者 map。

复杂度 O((n+q)n)O((n+q)\sqrt n)O((n+q)n​)。

const int mod=1e9+7;
int power(int x,int y=mod-2) {
	int sum=1;
	while (y) {
		if (y&1) sum=sum*x%mod;
		x=x*x%mod;y>>=1;
	}
	return sum;
}
int n,q,a[maxn],deep[maxn],fa[maxn];
int f[maxn][505],t[maxn],id[maxn];
const int base=500;
int dfs(int x,int y) {
	if (!x) return 0;
	if (t[deep[x]]<=base) {
		if (f[x][id[y]]!=-1) return f[x][id[y]];
		else return f[x][id[y]]=f[y][id[x]]=a[x]*a[y]+dfs(fa[x],fa[y]);
	}
	else return a[x]*a[y]+dfs(fa[x],fa[y]);
}
signed main(void){
	int i,x,y;
	memset(f,-1,sizeof(f));
	read(n);read(q);
	for (i=1;i<=n;i++) read(a[i]);
	for (i=2;i<=n;i++) read(fa[i]);
	for (i=1;i<=n;i++) {
		deep[i]=deep[fa[i]]+1;
		id[i]=t[deep[i]]++;
	}
	while (q--) {
		read(x);read(y);
		printf("%lld\n",dfs(x,y));
	}
	return 0;
}
