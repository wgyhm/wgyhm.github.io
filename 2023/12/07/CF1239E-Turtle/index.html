总结

Solution

令第一行的前缀和为 pip_ipi​，第二行的后缀和为 sis_isi​，则种排列所有的贡献为：max⁡(pi+si)\max(p_i+s_i)max(pi​+si​)。

假设我们已经直到第一行和第二行的选的数的集合，那么一定是第一行从小到大排序，第二行从大到小排序。贪心显然。

再考虑转移点，也就是从哪里开始第一行转到第二行是可能对最大值产生影响的。

考虑每次移动转移点的增量 wi=ai−bi−1w_i=a_i-b_{i-1}wi​=ai​−bi−1​，那么答案可以写为：max⁡(a1+∑b+∑j=2iwj)\max(a_1+\sum b+\sum_{j=2}^i w_j)max(a1​+∑b+∑j=2i​wj​)。

再观察到 aaa 是不降的，bbb 是不升的，所以 www 一定不降。www 一开始可能为负，所以转移点一定再 i=1i=1i=1 或者 i=ni=ni=n。

题目也就转化为了选择把 2n2n2n 个数分为两个大小为 nnn 的集合，最小化 max⁡(a1+∑b,bn+∑a)\max(a_1+\sum b,b_n+\sum a)max(a1​+∑b,bn​+∑a)。而 a1,bna_1,b_na1​,bn​ 我们可以先不管考虑枚举，也可以猜测到因为这两个一定要选，肯定是把最小的两个数放到这两个数上。

题目又变成了选择 2n−22n-22n−2 个数分别两个大小为 n−1n-1n−1 的集合，最小化 max⁡(∑a,∑b)\max(\sum a,\sum b)max(∑a,∑b)。这里就可以直接 dp 了。而由于 s=∑a+∑bs=\sum a+\sum bs=∑a+∑b 是确定的，只需要找到一个集合可以大于 ⌈s2⌉\lceil \dfrac{s}{2}\rceil⌈2s​⌉ 的第一个数。可以直接 O(n2∑a)O(n^2\sum a)O(n2∑a) 暴力背包。

发现取值都是 010101，考虑 bitset 优化，复杂度为 O(n2∑aw)O(\dfrac{n^2\sum a}{w})O(wn2∑a​)。

Tp void read(T &x){
    int f=1;x=0;char c=getchar();
    while (c<'0'||c>'9') {if (c=='-') f=-1;c=getchar();}
    while (c>='0'&&c<='9') {x=x*10+c-'0';c=getchar();}
    x*=f;
}

const int mod=1e9+7;
int power(int x,int y=mod-2) {
	int sum=1;
	while (y) {
		if (y&1) sum=sum*x%mod;
		x=x*x%mod;y>>=1;
	}
	return sum;
}
int a[maxn],b[maxn],c[maxn],tot1,tot2;
bitset<2500005>f[51][26];
int n,N;
signed main(void){
    int i,j,k;
	read(n);
    for (i=1;i<=2*n;i++) read(a[i]);
    sort(a+1,a+1+2*n);
    reverse(a+1,a+1+2*n);
    int N=2*n-2,sum=0;
    f[0][0][0]=1;
    for (i=1;i<=N;i++) {
        sum+=a[i];
        for (j=0;j<=min(i,n-1);j++) {
            f[i][j]=f[i-1][j];
            if (j) f[i][j]|=(f[i-1][j-1]<<a[i]);
        }
    }
    int ans=0;
    for (i=(sum+1)/2;i<=sum;i++) if (f[N][n-1][i]) {ans=i;break;}
    i=N,j=n-1,k=ans;
    while (i) {
        if (f[i-1][j][k]) b[++tot1]=a[i],i--;
        else c[++tot2]=a[i],j--,k-=a[i],i--;
    }
    b[++tot1]=a[2*n-1];
    c[++tot2]=a[2*n];
    sort(b+1,b+1+tot1);
    sort(c+1,c+1+tot2);
    for (i=1;i<=tot1;i++) printf("%d ",b[i]);put();
    for (i=tot2;i>=1;i--) printf("%d ",c[i]);put();
	return 0;
}
