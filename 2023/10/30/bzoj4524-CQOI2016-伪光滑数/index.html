Solution

指导说是堆的套路板子题啊，可是完全不会！！！！

具体就是枚举 aka_kak​ 和 kkk，每次找到第 iii 大的数，然后微调一下使得可能包含 k+1k+1k+1 大。需要保证不漏。最好保证不重。

这题保证不重需要记录之前的最小的质数，保证每次调整质数枚举最多枚举到之前的最小质数。

Tp void read(T &x){
    int f=1;x=0;char c=getchar();
    while (c<'0'||c>'9') {if (c=='-') f=-1;c=getchar();}
    while (c>='0'&&c<='9') {x=x*10+c-'0';c=getchar();}
    x*=f;
}

	Ts void _debug(char* f,T x,Ar... y){while(*f!=',') cerr<<*f++;cerr<<'='<<x<<",";_debug(f+1,y...);}
	#define gdb(...) _debug((char*)#__VA_ARGS__,__VA_ARGS__)
}using namespace Debug;
#define fi first

const int mod=1e9+7;
int power(int x,int y=mod-2) {
	int sum=1;
	while (y) {
		if (y&1) sum=sum*x%mod;
		x=x*x%mod;y>>=1;
	}
	return sum;
}
int n,k;
int prime[105],cnt,vis[205];
struct yyy {
	int w,m,t,p;
	yyy (int a=0,int b=0,int c=0,int d=0) {
		w=a,m=b,t=c,p=d;
	}
	bool operator <(const yyy &tmp) const {
		return w<tmp.w;
	}
};
priority_queue<yyy>q;

signed main(void){
//	freopen("1.in","r",stdin);
	int i,j;
	read(n),read(k);
	vis[1]=1;
	for (i=2;i<=127;i++) {
		if (!vis[i]) {
			int tot=0;
			prime[++cnt]=i;
			for (j=i;j<=n;j*=i) ++tot,q.push(yyy(j,cnt-1,tot,i)); 
		}
		for (j=i;j<=127;j+=i) vis[j]=1;
	}
	k--;
	while (k--) {
		auto tmp=q.top();q.pop();
		if (tmp.t>1) {
			for (i=1;i<=tmp.m;i++) q.push(yyy(tmp.w/tmp.p*prime[i],i,tmp.t-1,tmp.p));
		}
	}
	printf("%lld",q.top().w);
	return 0;
}
