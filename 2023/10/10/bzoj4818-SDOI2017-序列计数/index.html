[SDOI2017] 序列计数

https://www.luogu.com.cn/problem/P3702

Solution

一种很简单的做法是全部的减去不包含质数的，用矩阵乘法维护，在这里不赘述，除去筛质数的复杂度是 O(p3log⁡n)O(p^3\log n)O(p3logn) 的。

观察我们的矩阵，发现是 w0,w1,..,wp−1w_0,w_1,..,w_{p-1}w0​,w1​,..,wp−1​ 每次向右移动一位，移动 ppp 次构成的矩阵。这个矩阵很特殊，再找性质。

回到最初始的 dp，fi,j×wk→fi+1,(j+k)%pf_{i,j}\times w_k \to f_{i+1,(j+k)\% p}fi,j​×wk​→fi+1,(j+k)%p​，可以看成 fi,j×f1,k→fi+1,(j+k)%pf_{i,j}\times f_{1,k} \to f_{i+1,(j+k)\% p}fi,j​×f1,k​→fi+1,(j+k)%p​，进而，可以看成 fi,j×fk,l→fi+j,(k+l)%pf_{i,j}\times f_{k,l}\to f_{i+j,(k+l)\% p}fi,j​×fk,l​→fi+j,(k+l)%p​。

这个东西显然有结合律，就像矩阵一样。所以也可以快速幂。直接做是 O(p2log⁡n)O(p^2\log n)O(p2logn) 的。

而且这个是卷积的形式，也可以做到 O(plog⁡plog⁡n)O(p\log p\log n)O(plogplogn) 的复杂度。

int power(int x,int y=mod-2) {
	int sum=1;
	while (y) {
		if (y&1) sum=sum*x%mod;
		x=x*x%mod;y>>=1;
	}
	return sum;

int n,m,p;
int f[maxn],g[maxn],F[maxn],G[maxn];
bitset<20000005>vis;
int c[maxn],prime[5000005],cnt;
void mul(int *a,int *b) {
	int i,j;
	for (i=0;i<p;i++) c[i]=0;
	for (i=0;i<p;i++)
		for (j=0;j<p;j++)
			c[(i+j)%p]=(c[(i+j)%p]+a[i]*b[j])%mod;
	for (i=0;i<p;i++) a[i]=c[i];
}
signed main(void){
	int i,j;
	read(n);read(m);read(p);
	vis[1]=1;f[1]++,g[1]++;
	for (i=2;i<=m;i++) {
		f[i%p]++;
		if (!vis[i]) prime[++cnt]=i;
		else g[i%p]++;
		for (j=1;j<=cnt&&prime[j]*i<=m;j++) {
			vis[i*prime[j]]=1;
			if (i%prime[j]==0) break;
		}
//		gdb(i,vis[i]);

	while (n) {
		if (n&1) mul(F,f),mul(G,g);
		mul(f,f),mul(g,g),n>>=1;
	}

	return 0;
}
