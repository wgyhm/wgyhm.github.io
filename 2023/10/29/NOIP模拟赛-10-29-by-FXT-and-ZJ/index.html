T1

Description

给定 nnn，求满足 1≤x,y≤n,x2−y=k2,k∈N1\le x,y\le n,x^2-y=k^2,k\in N1≤x,y≤n,x2−y=k2,k∈N 的 x,yx,yx,y 对数。

n≤1012n\le 10^{12}n≤1012。

Solution

移项有 (x+k)(x−k)=y(x+k)(x-k)=y(x+k)(x−k)=y，枚举 x−kx-kx−k ，计算 x+kx+kx+k 的数量。注意两者要奇偶相同。二分也可以，直接算也可以。我用了二分，减少（大脑）计算量。

https://qoj.ac/contest/1382/problem/7566

Solution

第一步差分没想到，题目做傻了，大家警戒。

区间加，考虑差分。目标是 bib_ibi​ 都变成 000。

观察到可以减少一次操作当且仅当 ∑i∈Sbi=0\sum\limits_{i\in S} b_i=0i∈S∑​bi​=0。直接 dp 就行。

可以枚举子集，但发现很多余。

根据差分。？枚新加的哪个数就可以了。在 ∑bi=0\sum b_i=0∑bi​=0 的时候贡献 +1+1+1。

可能需要卡常。

#define lowbit(x) ((x)&(-(x)))
int f[(1<<23)+5],a[55],n;
ll s[(1<<23)+5],b[55];
void solve(void) {
	int i,j;
	read(n);
	for (i=1;i<=n;i++) read(a[i]);
	for (i=1;i<=n;i++) b[i]=a[i]-a[i-1],s[(1<<i-1)]=b[i];
	int N=(1<<n);
	for (i=1;i<N;i++) {
		f[i]=0;
		s[i]=s[i^lowbit(i)]+s[lowbit(i)];
		for (j=i;j;j-=lowbit(j)) f[i]=max(f[i],f[i^(lowbit(j))]);
		if (s[i]==0) f[i]++;
	}
	printf("%d\n",n-f[N-1]);
}
signed main(void){
	int T;
	read(T);
	while (T--) solve();
	return 0;
}

T3

Description

图片暂缺 · image-20231029162736654

n,q≤2×105n,q \le 2\times 10^5n,q≤2×105。

Solution

如果写过暴力的话，就发现满足条件的 x,yx,yx,y 一定有 x>yx>yx>y 且数量大约是 O(nlog⁡n)O(n\log n)O(nlogn) 级别的，想到做完所有答案最后输出。

观察到 GCD(x,y)=GCD(y,⌊xy⌋)\text{GCD}(x,y)=\text{GCD}(y,\left\lfloor\dfrac{x}{y}\right\rfloor)GCD(x,y)=GCD(y,⌊yx​⌋)，考虑枚举 yyy 和 ⌊xy⌋=z\left\lfloor\dfrac{x}{y}\right\rfloor=z⌊yx​⌋=z，GCD\text{GCD}GCD 函数最多只有 O(nln⁡n)O(n\ln n)O(nlnn) 种取值，对于每种 y,zy,zy,z ，算出 xxx 对应的范围，然后枚举 gcd⁡\gcdgcd，暴力判断。实测跑的很快。

复杂度大约 O(nlog⁡2n)O(n\log ^2 n)O(nlog2n)。

#define fi first

#define mk make_pair
const int mod=1e9+7;
int power(int x,int y=mod-2) {
	int sum=1;
	while (y) {
		if (y&1) sum=sum*x%mod;
		x=x*x%mod;y>>=1;
	}
	return sum;
}
int gcd(int x,int y){
    if(!y) return x;
    return gcd(y,x%y);
}
int GCD(int x,int y){
    if(!y) return x;
    return GCD(y,x/y);
}
int n,q;
pair<int,int> ans[5000005];
int cnt;
signed main(void){
	int i,j,x,y,z;
	read(n);read(q);
	for (y=2;y<=n;y++) {
		for (z=1;z<=n/y;z++) {
			int xl=y*z,xr=min(n,y*(z+1)-1),tmp=GCD(y,z);
//			gdb(y,z,xl,xr,tmp);
			if (y%tmp) continue;
			for (i=(xl-1)/tmp*tmp+tmp;i<=xr;i+=tmp) if (gcd(i,y)==tmp) ans[++cnt]=mk(i,y);   
		}
	} 
	sort(ans+1,ans+1+cnt);
	while (q--) {
		read(x);
		if (x>cnt) puts("-1 -1");
		else printf("%d %d\n",ans[x].fi,ans[x].se);
	}
	return 0;
}
