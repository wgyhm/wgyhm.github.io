Solution

这种算贡献的题目还是很不会啊。

考虑一个最终答案位置对答案的贡献，就是左右两边分割点出现时间的最大值，这样就把他隔开了，他也不会继续运行下去。

现在我们要计算点 iii 分割点出现的时间，也就是前 iii 个大小的数都前 iii 个位置。再观察冒泡排序的本质，如果如果存在排名前 iii 的数再 iii 之后，每次冒泡会将最右边的数向左边移动一格。这样的话直接计算即可。

具体的，排序即可。复杂度 O(nlog⁡n)O(n\log n)O(nlogn) ，瓶颈在排序。

#include<bits/stdc++.h>
#define int long long
#define ull unsigned long long
#define maxn 1000005
#define put() putchar('\n')
#define Tp template<typename Ty>
#define Ts template<typename Ty,typename... Ar>
using namespace std;
void read(int &x){
    int f=1;x=0;char c=getchar();
    while (c<'0'||c>'9') {if (c=='-') f=-1;c=getchar();}
    while (c>='0'&&c<='9') {x=x*10+c-'0';c=getchar();}
    x*=f;
}
namespace Debug{
	Tp void _debug(char* f,Ty t){cerr<<f<<'='<<t<<endl;}
	Ts void _debug(char* f,Ty x,Ar... y){while(*f!=',') cerr<<*f++;cerr<<'='<<x<<",";_debug(f+1,y...);}
	Tp ostream& operator<<(ostream& os,vector<Ty>& V){os<<"[";for(auto& vv:V) os<<vv<<",";os<<"]";return os;}
	#define gdb(...) _debug((char*)#__VA_ARGS__,__VA_ARGS__)
}using namespace Debug;
#define fi first
#define se second
#define mk make_pair
const int mod=1e9+7;
int power(int x,int y=mod-2) {
	int sum=1;
	while (y) {
		if (y&1) sum=sum*x%mod;
		x=x*x%mod;y>>=1;
	}
	return sum;
}
struct yyy {
	int x,id;
}a[maxn];
bool cmp(yyy x,yyy y) {return x.x==y.x?x.id<y.id:x.x<y.x;}
int n,ans,t[maxn];
signed main(void){
	int i;
	read(n);
	for (i=1;i<=n;i++) read(a[i].x),a[i].id=i;
	sort(a+1,a+1+n,cmp);
	int Max=0;
	for (i=1;i<=n;i++) {
		Max=max(Max,a[i].id);
		t[i]=Max-i;
	}
	for (i=1;i<=n;i++) {
		ans+=max(max(t[i],t[i-1]),1ll);
	}
	printf("%lld\n",ans);
	return 0;
}
