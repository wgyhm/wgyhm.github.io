Random Isolation

Description

给定一颗 nnn 个点的树，如果存在，则随机大于 kkk 的连通块，随机选择连通块中的一个点，删掉这个点以及相邻的边。期望操作次数。

k<n≤100k<n\le 100k<n≤100。2s。

Solution

根据期望的线性性，期望的操作次数为每个点期望的操作次数之和。考虑 O(n)O(n)O(n) 枚举以每个点为根。

考虑树的一颗子图，子图的节点数为 nnn，相邻的节点数为 mmm，则到达这个子图的概率为 1(n+mn)\dfrac{1}{\dbinom{n+m}{n}}(nn+m​)1​。所以一个子图的概率只与 nnn 和 mmm 相关，类似于树上背包 dp。

这样做是 O(n5)O(n^5)O(n5) 的，但观察到 n+m≤n+m\len+m≤ 总节点个数，所以常数比较小。能过。

const int mod=998244353;
int power(int x,int y=mod-2) {
	int sum=1;
	while (y) {
		if (y&1) sum=sum*x%mod;
		x=x*x%mod;y>>=1;
	}
	return sum;
}
int suf[maxn],isuf[maxn];
int C(int n,int m) {return suf[n]*isuf[m]%mod*isuf[n-m]%mod;}
int iC(int n,int m) {return suf[m]*suf[n-m]%mod*isuf[n]%mod;}
void add(int &x,int y) {x=(x+y)%mod;}
int n,m,ans,f[maxn][maxn][maxn],g[maxn][maxn],siz[maxn],inv[maxn];
vector<int>to[maxn];
void dfs(int x,int pre) {
	siz[x]=1;
	int i,j,k,l,cnt=0;
	f[x][1][0]=1;
	for (auto y:to[x]) if (y^pre) {
		dfs(y,x);
		memset(g,0,sizeof(g));
		for (i=1;i<=siz[x];i++)
			for (j=0;j<=siz[x]-i;j++) {
				add(g[i][j+1],f[x][i][j]);
				for (k=1;k<=siz[y];k++)
					for (l=0;l<=siz[y]-k;l++) {
						add(g[i+k][j+l],f[x][i][j]*f[y][k][l]);
					}	
			}
		siz[x]+=siz[y];
		for (i=0;i<=siz[x];i++) for (j=0;j<=siz[x];j++) f[x][i][j]=g[i][j];
	}
}
signed main(void){
	int i,j,k,x,y;double tmp=0;
	read(n);read(m);
	for (i=1;i<n;i++) {
		read(x);read(y);
		to[x].push_back(y);
		to[y].push_back(x);
	}
	for (inv[0]=1,i=1;i<=2*n;i++) inv[i]=power(i);
	for (suf[0]=1,i=1;i<=2*n;i++) suf[i]=suf[i-1]*i%mod;
	isuf[2*n]=power(suf[2*n]);
	for (i=2*n;i>=1;i--) isuf[i-1]=isuf[i]*i%mod;
	for (i=1;i<=n;i++) {
		memset(f,0,sizeof(f));
		dfs(i,0);
		for (j=m+1;j<=n;j++) for (k=0;k<=n-j;k++) {
			add(ans,iC(j+k,j)*f[i][j][k]%mod*inv[j]%mod);
		}
	}
	printf("%lld",ans);
	return 0;
}
