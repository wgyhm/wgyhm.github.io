P6811 「MCOI-02」Build Battle 建筑大师

Description

一个序列 nnn，给定独立询问 mim_imi​，序列为 1,2,3,...,mi,1,2,...,(n−1) mod mi+11,2,3,...,m_i,1,2,...,(n-1)\bmod m_i+11,2,3,...,mi​,1,2,...,(n−1)modmi​+1

求本质不同的子序列数。qqq 组询问。

n,q≤1×106n,q\le 1\times 10^6n,q≤1×106

1s 512MB

Solution

Step1

定义 fif_ifi​ 表示到 iii 的本质序列不同的个数。

枚举 xxx 表示子序列下一个权值，为了避免重复，nexi,xnex_{i,x}nexi,x​ 表示从 iii 开始，下一个权值为 xxx 的点的位置。fnex[i][x]⇐fif_{nex[i][x]}\Leftarrow f_ifnex[i][x]​⇐fi​。

复杂度 O(n2q)O(n^2q)O(n2q)

Step2

可以发现，这样丝毫没有用到这个序列的特殊性。

iii 可以从 i−1,i−2,...,i−mi-1,i-2,...,i-mi−1,i−2,...,i−m 转移过来。前缀和优化。

复杂度 O(nq)O(nq)O(nq)

Step3

令 si=∑j=1ifjs_i=\sum\limits_{j=1}^{i} f_jsi​=j=1∑i​fj​

则 si=si−1+fi=2si−1−si−m−1s_i=s_{i-1}+f_i=2s_{i-1}-s_{i-m-1}si​=si−1​+fi​=2si−1​−si−m−1​ 。我们要求的就是 sns_nsn​。

接下来是从题解贺来的小 trick。

考虑组合意义，相同于从 111 开始，目前在位置 iii ，可以走一步让 i+1i+1i+1 让阈值 v=v∗2v=v*2v=v∗2，也可以走到 i+m+1i+m+1i+m+1 让阈值 v=−vv=-vv=−v。答案就是走到位置 nnn 的阈值之和。

观察到和怎么走并没有关系，而和走的步数有关系。则枚举走 m+1m+1m+1 步的次数。则答案为

sn=∑i=0nm2n−(m+1)i×(−1)i×(n−(m+1)i+ii)s_n=\sum\limits_{i=0}^{\frac{n}{m}} 2^{n-(m+1)i} \times (-1)^i\times \dbinom{n-(m+1)i+i}{i}sn​=i=0∑mn​​2n−(m+1)i×(−1)i×(in−(m+1)i+i​)

每次枚举 mmm ，暴力计算。复杂度是调和级数 O(nlog⁡n)O(n\log n)O(nlogn)

#include<bits/stdc++.h>
#define int long long
#define ull unsigned long long
#define maxn 1000005
#define put() putchar('\n')
#define Tp template<typename Ty>
#define Ts template<typename Ty,typename... Ar>
using namespace std;
inline void read(int &x){
    int f=1;x=0;char c=getchar();
    while (c<'0'||c>'9') {if (c=='-') f=-1;c=getchar();}
    while (c>='0'&&c<='9') {x=x*10+c-'0';c=getchar();}
    x*=f;
}
namespace Debug{
	Tp void _debug(char* f,Ty t){cerr<<f<<'='<<t<<endl;}
	Ts void _debug(char* f,Ty x,Ar... y){while(*f!=',') cerr<<*f++;cerr<<'='<<x<<",";_debug(f+1,y...);}
	Tp ostream& operator<<(ostream& os,vector<Ty>& V){os<<"[";for(auto& vv:V) os<<vv<<",";os<<"]";return os;}
	#define gdb(...) _debug((char*)#__VA_ARGS__,__VA_ARGS__)
}using namespace Debug;
const int mod=1e9+7;
int suf[maxn*2],isuf[maxn*2],n,q,Ans[maxn],pw[maxn*2];
inline void add(int &x,int y) {x=(x+y)%mod;}
inline int power(int x,int y) {
	int ans=1;
	while (y) {
		if (y&1) ans=ans*x%mod;
		x=x*x%mod;y>>=1;
	}
	return ans;
}
inline int C(int x,int y) {return suf[x]*isuf[y]%mod*isuf[x-y]%mod;}
signed main(void){
	int i,m,j;
	read(n);read(q);
	suf[0]=1;pw[0]=1;
	for (i=1;i<=2*n;i++) pw[i]=pw[i-1]*2%mod,suf[i]=suf[i-1]*i%mod;
	isuf[2*n]=power(suf[2*n],mod-2);
	for (i=2*n;i>=1;i--) isuf[i-1]=isuf[i]*i%mod;
	for (j=1;j<=n;j++) {
		int m=j+1;
		for (i=0;i<=n/m;i++) {
			if (i&1) add(Ans[j],mod-pw[n-m*i]*C(n-m*i+i,i)%mod);
			else add(Ans[j],pw[n-m*i]*C(n-m*i+i,i));
		}
	} 
	while (q--) {
		read(m);
		printf("%lld\n",Ans[min(n,m)]);
	}
	return 0;
}
