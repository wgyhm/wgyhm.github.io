T1. ha

Description

图片暂缺 · image-20230712200555725

1≤n≤1×105,1≤m≤100,l1,l2,r1,r2≤m1\le n\le 1\times 10^5,1\le m\le 100,l1,l2,r1,r2\le m1≤n≤1×105,1≤m≤100,l1,l2,r1,r2≤m

Solution

这题虽然签到，但是重点在于 每次操作之后，都可以看到计数器当前的显示值。所以每一位是要单独计算的。

反着做，每次模拟 iii 这个位置两次不同效果的期望值大小。比较取大的即可。前缀和搞一下。

Code

			sum2/=(r2-l2+1);
			
			a[j]=max(sum1,sum2);
		}
	}

T2. carve

Description

图片暂缺 · image-20230712201016534

1≤n≤106,ai≤n1\le n\le 10^6,a_i\le n1≤n≤106,ai​≤n

Solution

实际上，题目转化为有两个机器，一个机器如果新加入或者加入和上一个加入的颜色不同，则花费一点代价。求代价最小化。

前几天做到一个这道题的加强版。就是有划分成 kkk 个有序集合。回到这题，考虑贪心。如果当前节点和原来机器的颜色都不一样，则贪心去机器下一个颜色一样的位置大的机器。证明略。具体看代码。

T3. sword

Description

给定 111 个长度为 nnn 的正整数序列 aaa。每次操作可以选择 aia_iai​ 加 111 或者减 111，但需要满足 aia_iai​ 为正数。

求让 gcd⁡(a1,a2,a3,...,an)>1\gcd(a_1,a_2,a_3,...,a_n)>1gcd(a1​,a2​,a3​,...,an​)>1 的最少操作次数。

n≤2×105,ai≤1×1012n\le 2\times 10^5,a_i\le 1\times 10^{12}n≤2×105,ai​≤1×1012

1s 512MB

Upt on 12.21：CF1305F

Solution

很有意思的题目！

一种显然的想法就是枚举质数 ppp ，如果答案为 ppp 的倍数的最小操作次数。这样做是 O(n)O(n)O(n) 的。

注意到 p=2p=2p=2 时，答案 ≤n\le n≤n。

考虑使枚举质数 ppp 的次数变少。观察到操作次数不超过 111 的个数有至少有 n2\dfrac{n}{2}2n​ 个。考虑随机出一个数 aia_iai​，枚举 ai−1,ai,ai+1a_i-1,a_i,a_i+1ai​−1,ai​,ai​+1 三个数的质因子。多随几个，保证正确率。随机 nnn 次的错误率是 (1/2)n(1/2)^n(1/2)n。

我的考场做法是枚举 [ai−100,ai+100][a_i-100,a_i+100][ai​−100,ai​+100] 的质因子。这样做效率显然是没有上面高的。但是能过/shl

	}

T4. war

Description

图片暂缺 · image-20230712202641404

n≤5×105n\le 5\times 10^5n≤5×105

同 CF798E

注意空间限制。

Solution

方心童说这个东西叫做 “线段树隐式建图”，反正是我没见过的。

空间为 O(n2)O(n^2)O(n2) 的暴力是显然的。

Step 1

方便起见，将等于 −1-1−1 的 aia_iai​ 改成 n+1n+1n+1。同时，记 pxp_xpx​ 为满足 ai=xa_i=xai​=x 的 iii （xxx 的前驱），如果没有就记为 n+1n+1n+1。

注意到其实对于所有 iii 满足：

  • fpi<fif_{p_i}<f_ifpi​​<fi​
  • 对于满足 j<aij<a_ij<ai​ 且 pj>ip_j>ipj​>i 的 jjj ，fj<fif_j<f_ifj​<fi​

注意到建出这两类有向边以后，跑拓扑排序。fif_ifi​ 即为其拓扑序。容易发现条件是充要的。

Step 2

有一个叫做 dfs 拓扑排序 的东西，过程如下：

  • 建反向边

  • 从 iii 进入，由 iii 出发的反向边，搜从没有经历过的点

  • 都搜完以后，时间戳 t⇐t+1t\Leftarrow t+1t⇐t+1，作为 iii 的拓扑序。

正确性是显然。每条边和每个点只会经过一次，复杂度上是 O(n+m)O(n+m)O(n+m) 的。

Step 3

考虑套用这个过程。

对于第一类边，只有 O(n)O(n)O(n) 条，直接跑。

对于第二类边，每次不断找 j<aij<a_ij<ai​ 中 pjp_jpj​ 最大的，如果 >i>i>i ，则继续搜。由于一个点只会经过一次，找到 jjj 以后删掉即可。

每个点只会删一次和经过一次，复杂度是 O(nlog⁡n)O(n\log n)O(nlogn) 的。

空间上，只有线段树和一些线性数组，甚至没有建边，空间复杂度是 O(n)O(n)O(n) 的。这可能也是 隐式建图 的来源。

inline void read(int &x){
    int f=1;x=0;char c=getchar();
    while (c<'0'||c>'9') {if (c=='-') f=-1;c=getchar();}
    while (c>='0'&&c<='9') {x=x*10+c-'0';c=getchar();}
    x*=f;
}
namespace Debug{
	Tp void _debug(char* f,Ty t){cerr<<f<<'='<<t<<endl;}
	Ts void _debug(char* f,Ty x,Ar... y){while(*f!=',') cerr<<*f++;cerr<<'='<<x<<",";_debug(f+1,y...);}
	Tp ostream& operator<<(ostream& os,vector<Ty>& V){os<<"[";for(auto& vv:V) os<<vv<<",";os<<"]";return os;}
	#define gdb(...) _debug((char*)#__VA_ARGS__,__VA_ARGS__)
}using namespace Debug;
int n,b[maxn];
int in[maxn],Ans[maxn],times;
int pre[maxn],f[maxn<<2];
inline void Pushup(int rt) {
	if (!f[rt<<1]||!f[rt<<1|1]) f[rt]=f[rt<<1]+f[rt<<1|1];
	else {
		if (pre[f[rt<<1]]>pre[f[rt<<1|1]]) f[rt]=f[rt<<1];
		else f[rt]=f[rt<<1|1]; 
	}  
}
inline void Update(int l,int r,int rt,int head,int k) {
	if (l==r) return f[rt]=k,void();
	int mid=l+r>>1;
	if (head<=mid) Update(l,mid,rt<<1,head,k);
	else Update(mid+1,r,rt<<1|1,head,k);
	Pushup(rt);
}
inline int Query(int l,int r,int rt,int head,int tail) {
	if (head<=l&&r<=tail) return f[rt];
	int mid=l+r>>1,tmp1=0,tmp2=0;
	if (head<=mid) tmp1=Query(l,mid,rt<<1,head,tail);
	if (tail>mid)  tmp2=Query(mid+1,r,rt<<1|1,head,tail);
	if (tmp1&&tmp2) {
		if (pre[tmp1]>pre[tmp2]) return tmp1;
		else return tmp2;
	}
	else return tmp1+tmp2;
}
int vis[maxn];
inline void dfs(int x) {
	if (Ans[x]||vis[x]) return ;
	int tmp;vis[x]=1;
	if (pre[x]<=n) dfs(pre[x]);
	if (b[x]>1) {
		while ((tmp=Query(1,n,1,1,b[x]-1))&&pre[tmp]>x) {
			Update(1,n,1,tmp,0);
			dfs(tmp);
		}	
	}
	Ans[x]=++times;
}

	for (i=1;i<=n;i++) {
		read(b[i]);
		if (b[i]!=-1) pre[b[i]]=i;else b[i]=n+1;
	}
	for (i=1;i<=n;i++) Update(1,n,1,i,i);
	gdb(n);
	for (i=1;i<=n;i++) if (!Ans[i]) dfs(i); 
	for (i=1;i<=n;i++) printf("%d ",Ans[i]);
	return 0;
}
