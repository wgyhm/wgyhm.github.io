T1

从大到小枚举因数，类似于埃式筛。

#include<bits/stdc++.h>
#define maxn 1000005
#define ll long long
#define put() putchar('\n')
using namespace std;
inline void read(int &x){
	int f=1;x=0;char c=getchar();
	while (c<'0'||c>'9') {if (c=='-') f=-1;c=getchar();}
	while (c>='0'&&c<='9') {x=x*10+c-'0';c=getchar();}
	x*=f;
}
int n;
int fa[maxn];
ll ans;
inline int gcd(int x,int y) {return y==0?x:gcd(y,x%y);}
inline int getfa(int x) {return x==fa[x]?x:fa[x]=getfa(fa[x]);}
inline void merge(int x,int y) {
	int fx,fy;
    fx=getfa(x);fy=getfa(y);
    if (fx^fy) fa[fx]=fy,ans+=gcd(x,y);
}
signed main(void){
//	freopen("1.in","r",stdin);
	int i,j;
	read(n);
	for (i=1;i<=n;i++) fa[i]=i;
	for (i=n;i>=1;i--) 
		for (j=i*2;j<=n;j+=i)
		    merge(i,j);
	printf("%lld",ans);
 	return 0;
}

#include<bits/stdc++.h>
#define maxn 500005
#define int long long
#define put() putchar('\n')
using namespace std;
inline void read(int &x){
	int f=1;x=0;char c=getchar();
	while (c<'0'||c>'9') {if (c=='-') f=-1;c=getchar();}
	while (c>='0'&&c<='9') {x=x*10+c-'0';c=getchar();}
	x*=f;
}
int n,m;
int a[maxn];
signed main(void){
//	freopen("1.in","r",stdin);
	int i,ans=0;
	read(n);read(m);
	for (i=1;i<=n;i++) read(a[i]);
	sort(a+1,a+1+n);
	int l=1,r=n;
	for (l=1;l<=n;l++) {
		if (l>=r) break;
		if (a[l]+a[r]>=m) r--,ans++;
		else continue;
	}
	printf("%lld",ans);
	return 0;
}
