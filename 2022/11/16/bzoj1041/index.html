Solution

求 x2+y2=r2x^2+y^2=r^2x2+y2=r2 的个数。

移项，y2=(r−x)(r+x)y^2=(r-x)(r+x)y2=(r−x)(r+x)

令 (r−x,r+x)=d(r-x,r+x)=d(r−x,r+x)=d ，即 r−x=du,r+x=dv,(u,v)=1r-x=du,r+x=dv,(u,v)=1r−x=du,r+x=dv,(u,v)=1

所以 y2=d2⋅uvy^2=d^2\cdot uvy2=d2⋅uv

而 (u,v)=1(u,v)=1(u,v)=1 ，所以 u,vu,vu,v 分别都为完全平方数。

令 u=s2,v=t2u=s^2,v=t^2u=s2,v=t2

所以 x=t2−s22⋅d,t=dst,r=t2+s22⋅dx=\dfrac{t^2-s^2}{2}\cdot d,t=dst,r=\dfrac{t^2+s^2}{2}\cdot dx=2t2−s2​⋅d,t=dst,r=2t2+s2​⋅d

枚举 2r2r2r 的因数 ddd，再枚举 sss 或者 ttt，就可以了。

inline void read(int &x){
    int f=1;x=0;char c=getchar();
    while (c<'0'||c>'9') {if (c=='-') f=-1;c=getchar();}
    while (c>='0'&&c<='9') {x=x*10+c-'0';c=getchar();}

namespace Debug{
	Tp void _debug(char* f,Ty t){cerr<<f<<'='<<t<<endl;}
	Ts void _debug(char* f,Ty x,Ar... y){while(*f!=',') cerr<<*f++;cerr<<'='<<x<<",";_debug(f+1,y...);}
	Tp ostream& operator<<(ostream& os,vector<Ty>& V){os<<"[";for(auto& vv:V) os<<vv<<",";os<<"]";return os;}
	#define gdb(...) _debug((char*)#__VA_ARGS__,__VA_ARGS__)
}using namespace Debug;
int R,n,ans; 
inline void solve(int d) {
	int now=n/d,i;
	for (i=1;i*i<=now;i++) {
		int sqt=sqrt(now-i*i);
		if (sqt*sqt+i*i==now&&__gcd(i,sqt)==1) {
			if ((sqt*sqt-i*i)%2) continue;//必须是2的倍数
			int x=(sqt*sqt-i*i)/2*d;
			int y=d*i*sqt;
			if (x>0&&y>0&&x*x+y*y==R*R) ans+=2;
		}
	}
}
signed main(void){
	int i,j,k;
	read(R);n=2*R;
	for (i=1;i*i<n;i++)
		if (n%i==0) {
			solve(i);solve(n/i);
		}
	if (i*i==n) solve(i);
	printf("%lld",(ans+1)*4);
	return 0;
}
